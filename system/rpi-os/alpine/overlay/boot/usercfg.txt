# Ajustes de vídeo/áudio do usuário para o Raspberry Pi 4B.
# O config.txt gerado pelo build faz "include usercfg.txt", então mantenha aqui
# apenas os overrides do produto (o config.txt base cuida de kernel/initramfs).
#
# ATENÇÃO: os valores do LCD Waveshare 4,3" HDMI (B) precisam ser confirmados no
# hardware — ver docs/waveshare/README.md. Comece por KMS; se o LCD não sincronizar,
# troque para o bloco fbdev/HDMI custom comentado no fim.

# Áudio: habilita a placa analógica bcm2835 do jack 3,5 mm, que É a saída de som
# do produto — o fone liga nela pelo cabo TRS interno do gabinete. /etc/asound.conf
# fixa essa placa pelo nome ("Headphones"). Sem esta linha a placa não existe e
# sobra só o HDMI, que se cala junto com o vídeo no Ctrl + AC (RF-04, WRN-013).
# TEM de vir ANTES do dtoverlay=vc4-kms-v3d: um dtparam depois de um dtoverlay vai
# primeiro para os parâmetros desse overlay, e o vc4-kms-v3d tem um "audio" próprio
# (áudio do HDMI0). Depois dele, esta linha liga o HDMI e a placa Headphones some.
dtparam=audio=on

# Driver gráfico KMS do Pi 4 (usado pelo Xorg via modesetting/mesa-dri-gallium).
# Também é o que dá hotplug de HDMI: o KMS atualiza /sys/class/drm em tempo de
# execução, que é como o app percebe o monitor externo ligado com a calculadora
# já em uso (RF-09). No caminho legado/firmware o modo é fixado no boot e essa
# detecção deixa de funcionar — ver README.md, "Monitor ligado com a calculadora
# já em uso".
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Sem borda preta em volta da imagem no LCD.
disable_overscan=1

# I2C1 nos pinos 3/5 (GPIO2 SDA, GPIO3 SCL): é o barramento onde o UPS HAT lê
# tensão/corrente/capacidade no endereço 0x42 (PRD §12, RF-06/RF-14). Sem isto
# /dev/i2c-1 não existe e a leitura de bateria não tem por onde acontecer.
# NÃO conflita com o teclado: pinout.md §6.3 mantém GPIO2/GPIO3 fora da matriz
# justamente para este barramento. Confira no hardware com `i2cdetect -y 1`.
dtparam=i2c_arm=on
